CADASTRO:

async function cadastrarChamado() {
    const formulario = document.querySelector("#formChamado");
    if (!formulario.reportValidity()) {
        return;
    }
    const novoChamado = {
        titulo: document.querySelector("#titulo").value.trim(),
        solicitante: document.querySelector("#solicitante").value.trim(),
        email: document.querySelector("#email").value.trim(),
        categoria: document.querySelector("#categoria").value,
        prioridade: document.querySelector("#prioridade").value,
        descricao: document.querySelector("#descricao").value.trim(),
        status: "Aberto"
    };
    try {
        await criarChamado(novoChamado);
        fecharModal();
        await carregarChamados();
        mostrarToast(
            "Chamado cadastrado com sucesso.",
            "sucesso"
        );
    }
    catch (erro) {
        console.error(erro);
        mostrarToast(
            "Não foi possível cadastrar o chamado.",
            "erro"
        );
    }
}

async function criarChamado(chamado) {
    const resposta = await fetch(API_URL, {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify(chamado)
    });
    const dados = await resposta.json();
    if (!resposta.ok) {
        throw new Error(dados.erro);
    }
    return dados;
}
